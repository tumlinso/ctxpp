cmake_minimum_required(VERSION 3.20)
project(ctxpp_core VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(ctxpp-core src/main.cpp src/clang_runtime.cpp)
target_include_directories(ctxpp-core PRIVATE include)
target_compile_options(ctxpp-core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ctxpp-core PRIVATE ${CMAKE_DL_LIBS})

option(CTXPP_BUILD_LIBTOOLING_BACKEND "Build the LibTooling/AST-matcher backend when Clang development files exist" ON)
if(CTXPP_BUILD_LIBTOOLING_BACKEND)
  find_package(LLVM CONFIG QUIET)
  find_package(Clang CONFIG QUIET)
  if(LLVM_FOUND AND Clang_FOUND)
    add_executable(ctxpp-libtooling-core src/libtooling_main.cpp)
    target_include_directories(ctxpp-libtooling-core PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
    target_compile_definitions(ctxpp-libtooling-core PRIVATE ${LLVM_DEFINITIONS})
    target_compile_options(ctxpp-libtooling-core PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(ctxpp-libtooling-core PRIVATE clangTooling clangASTMatchers clangIndex clangFrontend clangLex clangBasic)
    message(STATUS "ctxpp: LibTooling backend enabled")
  else()
    message(STATUS "ctxpp: Clang development package absent; building runtime libclang backend only")
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_test(NAME ctxpp_core_help COMMAND ctxpp-core --help)
endif()
